## 🎨 界面统一规范

### Modal 视图标准结构

```rust
pub struct XxxView {
    // 状态字段
    selected: usize,
    editing: bool,
}

impl ModalView for XxxView {
    fn kind(&self) -> ModalKind {
        ModalKind::Xxx
    }

    fn as_any_mut(&mut self) -> &mut dyn std::any::Any {
        self
    }

    fn handle_key(&mut self, key: KeyEvent) -> ViewAction {
        match key.code {
            KeyCode::Esc | KeyCode::Char('q') => ViewAction::Close,
            KeyCode::Up | KeyCode::Down => { self.toggle(); ViewAction::None },
            KeyCode::Enter => self.execute(),
            _ => ViewAction::None,
        }
    }

    fn render(&self, area: Rect, buf: &mut Buffer) {
        // 1. Clear
        Clear.render(popup_area, buf);
        
        // 2. 计算弹窗位置
        let popup_area = Rect { ... };
        
        // 3. 构建内容
        let lines = vec![...];
        
        // 4. 渲染
        let view = Paragraph::new(lines)
            .block(
                Block::default()
                    .title(" Title ")
                    .title_bottom(" Hints ")
                    .borders(Borders::ALL)
                    .border_style(...)
                    .style(...)
            );
        
        view.render(popup_area, buf);
    }
}
```
