### 2. 稳定性检测机制

```rust
// 连续 6 次内容不变 → 判定完成
let mut stable_count: usize = 0;
let stable_threshold: usize = 6;  // 6 * 2s = 12s 稳定

while elapsed < max_wait {
    if text != last_text {
        stable_count = 0;  // 重置计数器
    } else {
        stable_count += 1;
        if stable_count >= stable_threshold {
            return Ok(text);  // 回复完成
        }
    }
}
```

**优势：**
- ✅ 自动识别流式输出完成
- ✅ 避免过早截取不完整回复
- ✅ 避免无限等待已完成的回复

### 4. JS 直接操作 DOM

```rust
fn fill_input_via_js(&self, tab: &Tab, message: &str) -> Result<(), Box<dyn std::error::Error>> {
    // 清空输入框
    let clear_js = r#"(function() {
        const selectors = [
            'div[contenteditable="true"]',
            'textarea',
            'input[type="text"]'
        ];
        for (const sel of selectors) {
            const input = document.querySelector(sel);
            if (input) {
                input.focus();
                input.innerHTML = '';
                input.innerText = '';
                return 'cleared';
            }
        }
        return 'not-found';
    })()"#;

    tab.evaluate(clear_js, false)?;

    // 输入消息
    let type_js = format!(r#"(function() {{
        const input = document.querySelector('div[contenteditable="true"]');
        if (input) {{
            input.focus();
            input.innerText = '{}';
            input.dispatchEvent(new InputEvent('input', {{
                bubbles: true,
                inputType: 'insertText'
            }}));
            return 'typed';
        }}
        return 'not-found';
    }})()"#, escaped_message);

    tab.evaluate(&type_js, false)?;
}
```

**优势：**
- ✅ 避免焦点问题
- ✅ 更可靠的输入
- ✅ 备选 `type_into` 降级

```rust
fn close_popups(&self, tab: &Tab) -> Result<(), Box<dyn std::error::Error>> {
    let close_js = r#"(function() {
        const closeSelectors = [
            '.modal-close', '.close-btn', '[class*="close"]',
            '[aria-label="关闭"]', '[aria-label="Close"]',
            '.login-modal button', '[class*="guide"] button',
            '.welcome-modal button', '.tips-dialog button'
        ];

        for (const sel of closeSelectors) {
            const btn = document.querySelector(sel);
            if (btn && btn.offsetParent !== null) {
                btn.click();
                return 'closed:' + sel;
            }
        }
        return 'no-popup';
    })()"#;

    tab.evaluate(close_js, false)?;
}
```

```rust
use crate::client::DoubaoClient;

pub struct WebLlmClient {
    platform: WebLlmPlatform,
    browser: Option<Browser>,
    doubao_client: Option<DoubaoClient>,  // 豆包专用客户端
}

impl WebLlmClient {
    /// 创建豆包专用客户端
    pub fn new_doubao() -> Result<Self, Box<dyn std::error::Error>> {
        let doubao_client = DoubaoClient::new()?;
        Ok(Self {
            platform: WebLlmPlatform::Doubao,
            browser: None,
            doubao_client: Some(doubao_client),
        })
    }

    /// 发送消息
    pub fn send_message(&self, message: &str, ...) -> Result<String, ...> {
        match self.platform {
            WebLlmPlatform::Doubao => {
                // 优先使用专用客户端
                if let Some(ref client) = self.doubao_client {
                    client.chat(message)
                } else {
                    // 降级到浏览器自动化
                    self.send_message_via_browser(message, ...)
                }
            }
            WebLlmPlatform::Qianwen => {
                self.send_message_via_browser(message, ...)
            }
        }
    }
}
```
